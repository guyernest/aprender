// rng.rs — every random decision in the crate flows through these two functions
pub struct DomainKey(pub [u32; 2]);
/// key = trunc64(SHA-256(domain_tag ‖ root_seed_le ‖ domain_string)) — Pattern 2
pub fn derive_key(root_seed: u64, domain: &str) -> DomainKey;
/// counter = [ordinal_lo, ordinal_hi, stream_id, 0]; pure function of its arguments
pub fn draw(key: &DomainKey, stream_id: u32, ordinal: u64) -> [u32; 4];
/// 64-bit multiply-shift bounded draw from a 64-bit lane: ((x as u128 * n as u128) >> 64) as u64
pub fn bounded(key: &DomainKey, stream_id: u32, ordinal: u64, n: u64) -> u64;

// buckets.rs
pub struct ClassBuckets { /* BTreeMap<usize /*label*/, Vec<String /*id*/>> — each Vec SORTED */ }
impl ClassBuckets {
    pub fn from_split(train: &Split<Train>, exclusions: &ExclusionRecord) -> Self;
    pub fn class_sizes(&self) -> Vec<(usize, u64)>;
}

// select.rs
pub struct SelectionConfig { pub root_seed: u64, pub shots_per_class: u32 /* validated 8|16|32|64 */ }
pub struct Selection { /* ordered ids, per-row exact+normalized hash, class sizes, seed, shots */ }
impl FewShotSelector {
    pub fn select(
        train: &Split<Train>,
        validation_witness: &Split<Validation>,   // D-19: compat profile cannot supply this
        exclusions: &ExclusionRecord,
        cfg: &SelectionConfig,
        ledger: &mut AccessLedger,
    ) -> Result<Selection, ContrastiveDataError>;
}
impl Selection {
    pub fn ordered_ids(&self) -> &[String];
    pub fn semantic_hash(&self) -> [u8; 32];
    pub fn class_sizes(&self) -> &[(usize, u64)];
    pub fn contains(&self, id: &str) -> bool;   // pair-endpoint membership gate uses this

    /// Replay/verify constructor (D-16 untrusted-bytes boundary): rebuild a Selection from a
    /// replayed manifest against the ingested train split. Validates: every manifest ID exists
    /// in the split (else EndpointNotInSelection { id, found_in }); each row's exact + normalized
    /// hash matches the manifest's recorded hashes (else RowHashMismatch { id, expected, got });
    /// rebuilt semantic_hash must equal the manifest's recorded hash. 02-09's run_pairs calls
    /// THIS — the CLI never reconstitutes a Selection itself.
    pub fn from_manifest(manifest: &SelectionManifest, train: &Split<Train>)
        -> Result<Selection, ContrastiveDataError>;
}

// manifest.rs
pub struct SelectionManifest { /* schema_version, dataset fingerprint, seed, shots, ordered ids
    + hashes, excluded_ids + reduced pools, semantic_hash; volatile metadata OUTSIDE hashed region */ }
impl SelectionManifest {
    pub fn from_selection(sel: &Selection, excl: &ExclusionRecord, fp: &DatasetFingerprint) -> Self;
    pub fn to_canonical_bytes(&self) -> Vec<u8>;  // hashed region ONLY: BTreeMap order, no timestamps
    /// THE on-disk format of selection-manifest.json: canonical payload + an UNHASHED volatile
    /// wrapper (created_at, tool version). The crate owns this format; 02-09's CLI writes these
    /// bytes verbatim and never composes JSON around them.
    pub fn to_file_bytes(&self) -> Vec<u8>;
    /// Parses the FULL file form (wrapper + canonical payload). Volatile wrapper fields never
    /// affect hashed fields or semantic_hash.
    pub fn from_bytes(bytes: &[u8]) -> Result<Self, ContrastiveDataError>;
}
