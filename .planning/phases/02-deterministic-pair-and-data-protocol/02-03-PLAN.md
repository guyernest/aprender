```rust
// schema.rs — field NAMES and ORDER must match the committed StanceSample baseline
// (id, input, label, label_text, source_split) for byte-compatible JSONL round-trips.
#[derive(Debug, Clone, Serialize, Deserialize)]
#[serde(deny_unknown_fields)]
pub struct LabeledExample {
    pub id: String,
    pub input: String,
    pub label: usize,
    pub label_text: String,
    pub source_split: String,
}

// hash.rs
pub const CONTENT_NORMALIZATION_VERSION: &str = "nfc-trim-ws-v1";
pub fn exact_hash(input: &str) -> [u8; 32];
pub fn normalized_hash(input: &str) -> [u8; 32];
pub fn hex(digest: &[u8; 32]) -> String;

// split.rs — ZST roles; Split fields PRIVATE; sole constructor is the boundary
pub struct Train; pub struct Validation; pub struct Test; pub struct CompatibilityTest;
pub struct Split<R> { /* rows, fingerprint, PhantomData<R> */ }
pub struct SplitDeclaration { /* role name, expected per-class counts, label names, profile */ }
impl<R: SplitRole> Split<R> {
    pub fn from_jsonl_bytes(bytes: &[u8], decl: &SplitDeclaration)
        -> Result<(SourceHash, Split<R>), ContrastiveDataError>;
    pub fn rows(&self) -> &[LabeledExample];
    pub fn fingerprint(&self) -> &DatasetFingerprint;
}

// dedup.rs
pub struct ExclusionRecord { /* excluded_ids: Vec<String> (sorted), groups, reduced per-class pools */ }
pub fn cross_split_exclusions(
    train: &Split<Train>, others: &[&[LabeledExample]],
) -> ExclusionRecord;   // deterministic; never fails; underflow error surfaces at selection

// ledger.rs
pub struct AccessLedger { /* append-only Vec<AccessRecord> */ }
pub struct AccessRecord { /* role: &'static str, profile: String, purpose: String, fingerprint hex */ }
```

| Boundary | Description |
|----------|-------------|
| untrusted bytes → typed splits | JSONL bytes from disk/object storage claim a split role; must be validated, not trusted |
| train pool ↔ evaluation splits | Duplicate content silently crossing this line is evaluation leakage |
