// pairs.rs
pub struct CanonicalPair { /* lo, hi PRIVATE */ }
impl CanonicalPair {
    /// The ONLY constructor. Equal endpoints are Err(SelfPair) — no config resurrects (x,x).
    pub fn new(a: SelectedId, b: SelectedId) -> Result<Self, ContrastiveDataError>;
    pub fn lo(&self) -> &SelectedId; pub fn hi(&self) -> &SelectedId;
}
pub struct LabeledPair { pub pair: CanonicalPair, pub target: f32 /* 1.0 same / 0.0 diff, derived */ }

pub fn positive_capacity(class_sizes: &[u64]) -> u64;   // Σ C(n_k,2), checked arithmetic
pub fn negative_capacity(class_sizes: &[u64]) -> u64;   // Σ_{j<k} n_j·n_k
pub fn default_epoch_budget(class_sizes: &[u64]) -> u64; // 2·max(pos, neg)

#[non_exhaustive] pub enum PairStrategy { Oversampling }          // strategy_version: 1
#[non_exhaustive] pub enum SingletonPolicy { NegativesOnly }      // singleton_policy_version: 1

pub struct PairConfig { pub root_seed: u64, pub strategy: PairStrategy,
    pub singleton_policy: SingletonPolicy, pub budget: Option<u64> /* None → default */ }

pub struct PairSampler { /* buckets by class from Selection, prefix sums, key material — O(examples) */ }
impl PairSampler {
    pub fn new(sel: &Selection, cfg: &PairConfig) -> Result<Self, ContrastiveDataError>;
    /// Pure-indexed: pair_at(i) for i in 0..budget; iterator sugar over it.
    pub fn pair_at(&self, ordinal: u64) -> LabeledPair;
    pub fn iter_from(&self, offset: u64) -> impl Iterator<Item = LabeledPair> + '_;
    pub fn budget(&self) -> u64;
    #[cfg(test)] pub fn retained_state_size(&self) -> usize;  // structural, for 02-08's gate
}
/// Endpoint-membership gate for UNTRUSTED replayed pair bytes (the leaky negative hits this).

    singleton_policy + version, budget, deviation: [String; 3], affected_singleton_classes: u64 */ }
pub fn pair_manifest_hash(sampler: &PairSampler) -> [u8; 32]; // STREAMED SHA-256 — never collects
pub fn dump_pairs<W: std::io::Write>(sampler: &PairSampler, w: W) -> Result<(), ContrastiveDataError>;
