pub struct SentenceBatch {
    pub input_ids: Vec<u32>,        // row-major [B*S]
    pub token_type_ids: Vec<u32>,   // row-major [B*S]
    pub attention_mask: Vec<u8>,    // row-major [B*S], 1 keep / 0 pad
    pub batch: usize,
    pub seq: usize,                 // padded length (longest in batch, <= 256)
    pub truncation: Vec<TruncationFact>,   // per input: { truncated: bool, original_len: usize }
    pub provenance: Vec<InputProvenance>,  // per input: { index: usize, text_sha256: String }
}
pub struct MiniLmTokenizer { /* wraps tokenizers::Tokenizer */ }
impl MiniLmTokenizer {
    pub fn from_bytes(bytes: &[u8]) -> Result<Self, SetFitError>;  // records sha256 of bytes
    pub fn tokenizer_sha256(&self) -> &str;
    pub fn encode_batch(&self, texts: &[&str]) -> Result<SentenceBatch, SetFitError>; // max_length 256
}

// setfit/import.rs (feature = "setfit")
pub struct MiniLmImport { /* validated config + AprV2Reader handle + revision + tokenizer hash */ }

    /// Full pin: config equality vs BertConfig::minilm_l6(), tokenizer sha, revision record.
